  <div id="score">Score: 0</div>

    <h2 id="question"></h2>

  <div id="answers"></div>

  <br>
  <button id="kickBtn" onclick="kick()" disabled>Kick Field Goal</button>

  <p id="message"></p>

<script>
  const questions = [
    { q: "Energy due to motion is called:", a: ["Potential Energy", "Kinetic Energy", "Thermal Energy"], correct: 1 },
    { q: "A moving car, bouncing ball, or running child are examples of:", a: ["Kinetic Energy", "Temperature", "Pressure"], correct: 0 },
    { q: "The measure of the average kinetic energy of particles is:", a: ["Heat", "Temperature", "Thermal Expansion"], correct: 1 },
    { q: "Stored energy due to interactions between objects or particles is:", a: ["Potential Energy", "Thermal Energy", "Chemical Formula"], correct: 0 },
    { q: "Hot water steaming and heated metal glowing red-hot are examples of:", a: ["Thermal Energy", "Kinetic Energy", "Atoms"], correct: 0 },
    { q: "A small particle that is the building block of matter is an:", a: ["Element", "Atom", "Molecule"], correct: 1 },
    { q: "Matter with a composition that is always the same is a:", a: ["Compound", "Substance", "Element"], correct: 1 },
    { q: "A substance that consists of only one type of atom is an:", a: ["Element", "Compound", "Molecule"], correct: 0 },
    { q: "A substance containing atoms of two or more elements chemically bonded is a:", a: ["Element", "Compound", "Atom"], correct: 1 },
    { q: "H₂O, CO₂, and NaCl are examples of:", a: ["Chemical Formulas", "Elements", "Pressure"], correct: 0 },
    { q: "A decrease in a material's volume when temperature decreases is:", a: ["Thermal Expansion", "Thermal Contraction", "Heat"], correct: 1 },
    { q: "An increase in a material's volume when temperature increases is:", a: ["Thermal Expansion", "Pressure", "Vaporization"], correct: 0 },
    { q: "The transfer of thermal energy from warmer to cooler areas is:", a: ["Heat", "Temperature", "Kinetic Energy"], correct: 0 },
    { q: "Two or more atoms held together by covalent bonds form a:", a: ["Atom", "Element", "Molecule"], correct: 2 }
  ];

  let current = 0;
  let canKick = false;
  let score = 0;

  function updateScore() {
    document.getElementById("score").innerText = "Score: " + score;
  }

  function loadQuestion() {
    const q = questions[current];
    document.getElementById("question").innerText = q.q;

    const answersDiv = document.getElementById("answers");
    answersDiv.innerHTML = "";

    q.a.forEach((ans, index) => {
      const btn = document.createElement("button");
      btn.innerText = ans;
      btn.style.margin = "10px";
      btn.onclick = () => checkAnswer(index);
      answersDiv.appendChild(btn);
    });
  }

  function checkAnswer(index) {
    const q = questions[current];
    if (index === q.correct) {
      document.getElementById("message").innerText =
        "Correct! You earned a field goal attempt!";
      score += 5;
      updateScore();
      canKick = true;
      document.getElementById("kickBtn").disabled = false;
    } else {
      document.getElementById("message").innerText =
        "Incorrect! No kick this round.";
      nextQuestion();
    }
  }

  function kick() {
    if (!canKick) return;

    const success = Math.random() < 0.7;
    if (success) {
      document.getElementById("message").innerText =
        "🏈 You made the field goal!";
      score += 3;
    } else {
      document.getElementById("message").innerText =
        "❌ You missed the field goal!";
    }

    updateScore();
    canKick = false;
    document.getElementById("kickBtn").disabled = true;
    nextQuestion();
  }

  function nextQuestion() {
    current++;
    if (current >= questions.length) {
      document.getElementById("question-box").innerHTML =
        "<h2>Game Over!</h2><h3>Your Final Score: " + score + "</h3>";
      document.getElementById("answers").innerHTML = "";
      document.getElementById("kickBtn").style.display = "none";
      document.getElementById("message").innerText = "";
      return;
    }
    loadQuestion();
  }

  loadQuestion();
  updateScore();
</script>
